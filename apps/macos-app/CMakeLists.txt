# Vpipe Manager -- a macOS app wrapping the two command-line binaries.
#
# The GUI is a LAUNCHER, not a second implementation: it drives
# Contents/Helpers/{vpipe,vpipe-web-ui} by argv, exactly as a user would
# from a terminal. Nothing about a pipeline run is reachable only from
# the app, and the CLI cannot drift out from under it.
#
# Built by swiftc directly rather than through an Xcode project, so the
# whole tree stays one `cmake --build`. Produces an AD-HOC signed bundle
# that runs locally; Developer ID signing, notarization and the .dmg are
# a separate release step.
#
#   cmake --build <build> --target macos-app
#
# Options (all off/empty by default -- a plain build is unchanged):
#   VPIPE_BUILD_MACOS_APP    build the bundle at all (default ON on
#                            Apple Silicon)
#   VPIPE_BUNDLE_FFMPEG      copy FFmpeg into the bundle
#   VPIPE_FFMPEG_BUNDLE_DIR  where to copy it from
#   VPIPE_APP_SIGN_IDENTITY  codesign identity ("-" = ad-hoc)
#   VPIPE_APP_BUNDLE_ID      reverse-DNS bundle identifier

if(NOT APPLE)
    return()
endif()

option(VPIPE_BUILD_MACOS_APP "Build the Vpipe Manager.app bundle" ON)
if(NOT VPIPE_BUILD_MACOS_APP)
    return()
endif()

find_program(VPIPE_SWIFTC swiftc)
if(NOT VPIPE_SWIFTC)
    message(STATUS "Vpipe Manager: swiftc not found; skipping the app")
    return()
endif()

set(VPIPE_APP_BUNDLE_ID "com.tgous.vpipe" CACHE STRING
    "Bundle identifier for Vpipe Manager.app")
set(VPIPE_APP_SIGN_IDENTITY "-" CACHE STRING
    "codesign identity for Vpipe Manager.app ('-' is ad-hoc)")
option(VPIPE_BUNDLE_FFMPEG
    "Copy FFmpeg dylibs into Vpipe Manager.app/Contents/Frameworks" OFF)
set(VPIPE_FFMPEG_BUNDLE_DIR "" CACHE PATH
    "Directory holding the libav*/libsw* dylibs to bundle")

# Sparkle auto-update. Optional and OFF unless a framework is supplied:
# it is a 15 MB binary dependency that only a signed, published build
# needs, so `fetch-sparkle.sh` gets it out-of-tree and this points at
# the result. Without it the app builds and runs, minus the updater.
set(VPIPE_SPARKLE_DIR "" CACHE PATH
    "Directory containing Sparkle.framework (see fetch-sparkle.sh)")
set(VPIPE_APP_FEED_URL
    "https://github.com/tgo-app-dev/vpipe/releases/latest/download/appcast.xml"
    CACHE STRING "Sparkle appcast URL")
# Public half of the EdDSA pair generated by Sparkle's generate_keys.
# The PRIVATE half lives in the release machine's keychain and is never
# in this tree.
set(VPIPE_APP_ED_PUBKEY "Y1vMShVjOT6BQ2xxZa2Y66f7hoy9nhQrgIjfZo86pSw="
    CACHE STRING "Sparkle EdDSA public key")

# CFBundleVersion must increase monotonically for Sparkle to recognise an
# update, and it must not depend on who is building. The commit count is
# both. A shallow clone or a source tarball has no history to count, so
# fall back rather than fail the configure.
execute_process(
    COMMAND git -C ${CMAKE_SOURCE_DIR} rev-list --count HEAD
    OUTPUT_VARIABLE VPIPE_APP_BUILD
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET)
if(NOT VPIPE_APP_BUILD)
    set(VPIPE_APP_BUILD 1)
endif()

configure_file(
    ${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in
    ${CMAKE_CURRENT_BINARY_DIR}/Info.plist
    @ONLY)

# ---------------------------------------------------------------------
# 1. the Swift executable
# ---------------------------------------------------------------------
file(GLOB VPIPE_APP_SWIFT_SOURCES CONFIGURE_DEPENDS
     ${CMAKE_CURRENT_SOURCE_DIR}/Sources/*.swift)
if(NOT VPIPE_APP_SWIFT_SOURCES)
    message(FATAL_ERROR "Vpipe Manager: no Swift sources under Sources/")
endif()

execute_process(
    COMMAND xcrun --sdk macosx --show-sdk-path
    OUTPUT_VARIABLE VPIPE_MACOS_SDK
    OUTPUT_STRIP_TRAILING_WHITESPACE)

set(VPIPE_APP_SWIFT_EXE ${CMAKE_CURRENT_BINARY_DIR}/vpipe-app)

# Sparkle is reached through a compile-time flag rather than a runtime
# check, because `import Sparkle` cannot even be parsed when the
# framework is absent. The Swift sources guard every reference with
# `#if VPIPE_SPARKLE`, so the same sources build both ways.
set(_sparkle_swift_flags "")
set(_sparkle_framework "")
if(VPIPE_SPARKLE_DIR)
    if(NOT EXISTS ${VPIPE_SPARKLE_DIR}/Sparkle.framework)
        message(FATAL_ERROR
            "VPIPE_SPARKLE_DIR=${VPIPE_SPARKLE_DIR} has no "
            "Sparkle.framework in it. Run "
            "apps/macos-app/fetch-sparkle.sh and pass the directory it "
            "prints.")
    endif()
    set(_sparkle_framework ${VPIPE_SPARKLE_DIR}/Sparkle.framework)
    # -rpath so the linked framework resolves from inside the bundle at
    # run time; -F/-I so swiftc finds its module at compile time.
    set(_sparkle_swift_flags
        -DVPIPE_SPARKLE
        -F ${VPIPE_SPARKLE_DIR}
        -Xlinker -rpath -Xlinker @executable_path/../Frameworks
        -framework Sparkle)
    message(STATUS "Vpipe Manager: Sparkle from ${VPIPE_SPARKLE_DIR}")
else()
    message(STATUS "Vpipe Manager: no Sparkle (auto-update disabled)")
endif()

# -parse-as-library: the @main App struct is the entry point. Without
# it swiftc treats a file named main-ish as a script and the attribute
# is rejected.
#
# The deployment target is 13.0 (Ventura): MenuBarExtra needs it, and it
# is old enough to cover every Mac that can run the Apple-Silicon
# inference path anyway.
add_custom_command(
    OUTPUT ${VPIPE_APP_SWIFT_EXE}
    COMMAND ${VPIPE_SWIFTC}
            -O
            -target arm64-apple-macos13.0
            -sdk ${VPIPE_MACOS_SDK}
            -parse-as-library
            -module-name VpipeApp
            ${_sparkle_swift_flags}
            ${VPIPE_APP_SWIFT_SOURCES}
            -o ${VPIPE_APP_SWIFT_EXE}
    DEPENDS ${VPIPE_APP_SWIFT_SOURCES}
    COMMENT "Compiling Vpipe Manager Swift sources"
    VERBATIM)

add_custom_target(vpipe-app-swift DEPENDS ${VPIPE_APP_SWIFT_EXE})

# ---------------------------------------------------------------------
# 2. resources staged for Contents/Resources
# ---------------------------------------------------------------------
# Deliberately EMPTY of pipelines.
#
# The ready-made .vpipeline files used to ship here, and that was wrong
# on both counts. The prepare-* ones are templates that have to be
# edited before they do anything (an hf_token, a work directory), and a
# file inside a signed bundle cannot be edited -- writing to it breaks
# the code signature and the change is lost on the next update. The
# generation ones belong under the work directory's sandbox/, where the
# web UI can see and load them. Users fetch them from the repo or the
# docs into the right folder; the app finds them there.
#
# What the directory DOES carry is the app icon and the licence
# notices.
set(VPIPE_APP_RES ${CMAKE_CURRENT_BINARY_DIR}/Resources)

# ---- license notices -------------------------------------------------
# The bundle carries the licence of everything binary inside it. FFmpeg
# and LAME are LGPL-2.1, which REQUIRES shipping a copy of the licence
# and a notice, so this is an obligation rather than a courtesy; OpenSSL
# (Apache-2.0) and Sparkle (MIT) require their notices retained.
#
# The texts are vendored under licenses/ rather than copied off the
# build machine: they must be present whatever happens to be installed,
# and a notice that silently goes missing is the failure mode that
# matters.
#
# Only what is ACTUALLY bundled is shipped. Declaring a dependency the
# binary does not contain is its own kind of wrong.
# DERIVED from the libraries actually being bundled, not configured.
#
# FFmpeg generates libavutil/ffversion.h with its own release string, so
# the notice can name the version it is really shipping. A hand-set
# variable cannot: this started life as a CACHE entry, and when the
# bundled build moved 7.1.1 -> 8.1.2 the cache kept the old value and
# the notice went on naming 7.1.1 while the bundle carried 8.1.2. A
# licence notice that misidentifies the version it covers is worse than
# no notice, and nothing in a build log would have flagged it.
#
# The cache entry survives only as an override for a layout where the
# headers are not beside the libraries.
set(VPIPE_FFMPEG_VERSION "" CACHE STRING
    "Override the bundled FFmpeg version named in the notice (normally \
detected from libavutil/ffversion.h)")

if(VPIPE_BUNDLE_FFMPEG AND VPIPE_FFMPEG_BUNDLE_DIR)
    set(_ffver "")
    # <prefix>/lib is the usual pointer, so the headers are ../include.
    foreach(_cand "${VPIPE_FFMPEG_BUNDLE_DIR}/../include"
                  "${VPIPE_FFMPEG_BUNDLE_DIR}/include")
        if(NOT _ffver AND EXISTS "${_cand}/libavutil/ffversion.h")
            file(STRINGS "${_cand}/libavutil/ffversion.h" _l
                 REGEX "^#define[ \t]+FFMPEG_VERSION[ \t]+\"")
            if(_l)
                string(REGEX MATCH "\"([^\"]+)\"" _m "${_l}")
                set(_ffver "${CMAKE_MATCH_1}")
            endif()
        endif()
    endforeach()
    if(_ffver AND NOT VPIPE_FFMPEG_VERSION)
        set(VPIPE_FFMPEG_VERSION "${_ffver}")
        message(STATUS "Vpipe Manager: bundling FFmpeg ${VPIPE_FFMPEG_VERSION} "
                       "(detected)")
    elseif(NOT VPIPE_FFMPEG_VERSION)
        # Refuse to guess. The version names the corresponding-source
        # tarball a user is entitled to fetch; a wrong one sends them to
        # a URL for different code.
        message(FATAL_ERROR
            "Cannot determine the bundled FFmpeg version: no "
            "libavutil/ffversion.h near ${VPIPE_FFMPEG_BUNDLE_DIR}. Set "
            "VPIPE_FFMPEG_VERSION explicitly -- it names the exact "
            "source the LGPL notice offers.")
    endif()
endif()
# A SINGLE PINNED tarball, not one per release: the build is
# reproducible from this exact source, so re-uploading it with every
# .dmg would only duplicate bytes. Must stay reachable for as long as
# any release that names it is distributed -- which is why it is
# mirrored here rather than pointing at ffmpeg.org, whose old releases
# are pruned.
#
# DERIVED from the version, for the same reason the version is derived
# from the libraries: as a cache entry with an interpolated default it
# was computed once and then kept pointing at 7.1.1 after the bundle
# moved to 8.1.2 -- offering source for a version that is not the one
# shipped, which is precisely the promise the notice must not get wrong.
# Set the cache entry to override the host; the filename always follows
# the detected version.
set(VPIPE_FFMPEG_SOURCE_BASE
    "https://github.com/tgo-app-dev/vpipe/releases/download"
    CACHE STRING "Base URL hosting the pinned FFmpeg source tarball")
set(VPIPE_FFMPEG_SOURCE_URL
    "${VPIPE_FFMPEG_SOURCE_BASE}/ffmpeg-source-${VPIPE_FFMPEG_VERSION}/ffmpeg-${VPIPE_FFMPEG_VERSION}.tar.xz")

# Re-stage whenever a licence text or the generated notice changes.
add_custom_command(
    OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/resources.stamp
    COMMAND ${CMAKE_COMMAND} -E rm -rf ${VPIPE_APP_RES}
    COMMAND ${CMAKE_COMMAND} -E make_directory ${VPIPE_APP_RES}/Licenses
    COMMAND ${CMAKE_COMMAND} -E copy
            ${VPIPE_APP_ICON} ${VPIPE_APP_RES}/AppIcon.icns
    COMMAND ${CMAKE_COMMAND} -E copy
            ${CMAKE_CURRENT_BINARY_DIR}/NOTICE.txt
            ${CMAKE_SOURCE_DIR}/THIRD_PARTY_LICENSES.md
            ${_lic_files}
            ${VPIPE_APP_RES}/Licenses/
    # The stamp lives OUTSIDE the staged directory: anything inside it
    # is copied into Contents/Resources, and a build artefact has no
    # business shipping.
    COMMAND ${CMAKE_COMMAND} -E touch
            ${CMAKE_CURRENT_BINARY_DIR}/resources.stamp
    DEPENDS ${_lic_files} ${CMAKE_CURRENT_BINARY_DIR}/NOTICE.txt
            ${CMAKE_SOURCE_DIR}/THIRD_PARTY_LICENSES.md ${VPIPE_APP_ICON}
    COMMENT "Staging Vpipe Manager icon + licences"
    VERBATIM)

# ---------------------------------------------------------------------
# 3. assemble + sign
# ---------------------------------------------------------------------
# The bundle is what the user drags into /Applications, so it
# carries the product name. The executable inside stays "vpipe";
# only the wrapper is renamed. Paths with a space flow through
# CMake (VERBATIM) and bundle-app.sh (every expansion quoted).
set(VPIPE_APP_BUNDLE "${CMAKE_BINARY_DIR}/Vpipe Manager.app")

set(_bundle_args
    --app          "${VPIPE_APP_BUNDLE}"
    --swift-exe    ${VPIPE_APP_SWIFT_EXE}
    --cli          $<TARGET_FILE:vpipe-cli>
    --webui        $<TARGET_FILE:vpipe-web-ui>
    # The SONAME file (libvpipe.0.dylib), not the versioned real file:
    # that is the name recorded in every dependent's load command.
    --libvpipe     $<TARGET_SONAME_FILE:vpipe>
    --plist        ${CMAKE_CURRENT_BINARY_DIR}/Info.plist
    --entitlements ${CMAKE_CURRENT_SOURCE_DIR}/vpipe.entitlements
    --resources    ${VPIPE_APP_RES}
    --sign-identity ${VPIPE_APP_SIGN_IDENTITY})

if(_sparkle_framework)
    list(APPEND _bundle_args --sparkle-framework ${_sparkle_framework})
endif()

if(VPIPE_BUNDLE_FFMPEG)
    if(NOT VPIPE_FFMPEG_BUNDLE_DIR)
        message(FATAL_ERROR
            "VPIPE_BUNDLE_FFMPEG is ON but VPIPE_FFMPEG_BUNDLE_DIR is "
            "empty. Point it at the lib/ directory of an LGPL FFmpeg "
            "build -- see apps/macos-app/build-lgpl-ffmpeg.sh. Do NOT "
            "point it at Homebrew's: that build is --enable-gpl, which "
            "is incompatible with shipping this Apache-2.0 project.")
    endif()
    list(APPEND _bundle_args --ffmpeg-dir ${VPIPE_FFMPEG_BUNDLE_DIR})
endif()

add_custom_target(macos-app
    COMMAND ${CMAKE_COMMAND} -E env bash
            ${CMAKE_CURRENT_SOURCE_DIR}/bundle-app.sh ${_bundle_args}
    DEPENDS ${VPIPE_APP_SWIFT_EXE} ${CMAKE_CURRENT_BINARY_DIR}/resources.stamp
            vpipe-cli vpipe-web-ui vpipe
    COMMENT "Assembling ${VPIPE_APP_BUNDLE}"
    VERBATIM)

message(STATUS "Vpipe Manager: ${VPIPE_APP_BUNDLE_ID} "
               "(build ${VPIPE_APP_BUILD}, ffmpeg "
               "bundled=${VPIPE_BUNDLE_FFMPEG})")
